euclidean = (a, b) => {
  let sum = 0;
  for (let i = 0; i < a.length; i++) {
    const difference = a[i] - b[i];
    sum += difference * difference;
  }
  return Math.sqrt(sum / a.length);
}

selectFlag = code => {
  const control = document.querySelector("#selected-flag");
  if (!control || !flagByCode.has(code)) return;
  control.value = code;
  control.dispatchEvent(new Event("input", { bubbles: true }));
}

galleryFlags = {
  const query = searchText.trim().toLocaleLowerCase();
  return flags
    .filter(flag =>
      (!query || `${flag.name} ${flag.code}`.toLocaleLowerCase().includes(query)) &&
      (continentFilter === "All continents" || flag.continent === continentFilter) &&
      (colorFilter === "All colors" || flag.color === colorFilter)
    )
    .sort((a, b) => {
      if (sortBy === "Continent") {
        return a.continent.localeCompare(b.continent) || a.name.localeCompare(b.name);
      }
      if (sortBy === "Color") {
        return a.color.localeCompare(b.color) || a.name.localeCompare(b.name);
      }
      return a.name.localeCompare(b.name);
    });
}

quizTarget = (gameRound, flags[Math.floor(Math.random() * flags.length)])

quizOptions = {
  const distractors = flags.filter(flag => flag.code !== quizTarget.code);
  const shuffled = distractors.sort(() => Math.random() - 0.5).slice(0, 3);
  return [...shuffled, quizTarget].sort(() => Math.random() - 0.5);
}

viewof gameGuess = Inputs.select(quizOptions.map(flag => flag.code), {
  label: "Which country?",

})

  reduce: () => ({ round: gameRound, guess: gameGuess })
})
