Flags

regionNames = new Intl.DisplayNames(["en"], { type: "region" })
ukNames = ({
  "gb-eng": "England",
  "gb-nir": "Northern Ireland",
  "gb-sct": "Scotland",
  "gb-wls": "Wales"
})

flagLabel = (code) => ukNames[code] ?? regionNames.of(code.toUpperCase())

euclidean = (a, b) => {
  let sum = 0;
  for (let i = 0; i < a.length; i++) {
    const d = a[i] - b[i];
    sum += d * d;
  }
  return Math.sqrt(sum / a.length);
}

flagDistance = (f1, f2, metric) => {
  const layoutDist = euclidean(f1.grid, f2.grid);
  const colorDist = euclidean(f1.palette, f2.palette) * 3.5;
  const shapeDist = Math.abs(f1.aspect - f2.aspect);

  if (metric === "Combined") return 0.55 * layoutDist + 0.35 * colorDist + 0.10 * shapeDist;
  if (metric === "Layout") return layoutDist;
  if (metric === "Color") return colorDist;
  return shapeDist + 0.01 * colorDist;
}

selectedObj = flag_data.find(d => d.code === selectedFlag)

rankedFlags = flag_data
  .filter(d => d.code !== selectedFlag)
  .map(d => ({
    ...d,
    name: flagLabel(d.code),
    dist: flagDistance(selectedObj, d, similarityMetric)
  }))
  .sort((a, b) => a.dist - b.dist)

closest3 = rankedFlags.slice(0, 3)
distant3 = rankedFlags.slice(-3).reverse()
viewof selectedFlag = Inputs.select(flag_data.map(d => d.code), {
  label: "Flag",
  format: flagLabel,
  sort: true,
  value: "us"
})

  ["Combined", "Layout", "Color", "Shape"],
  {
    label: "Match",
    value: "Combined"
  }
)
html`
